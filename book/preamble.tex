% LaTeX preamble for the PDF build.
%
% This lives in a .tex file rather than in metadata.yaml because pandoc parses
% YAML `header-includes` as Markdown: simple \usepackage lines survive, but
% comments become escaped \% literals and multi-line macro definitions get
% reflowed into prose. --include-in-header is passed through raw.
% It is also only applied to the PDF, where LaTeX means anything.

% --- title block ------------------------------------------------------------
% Stack the author entries vertically. Without this, multiple authors are set
% side by side and the attribution runs off the page.
\renewcommand{\and}{\\}

% Put the programme logo above the title. titling gives us a hook into
% \maketitle without having to carry a modified copy of pandoc's template.
%
% The RGB file, per the 2021 brand guidelines: RGB for screen, CMYK for print at
% 300 dpi or better. This document is read on screen far more often than it is
% printed, and at the width set below the RGB file still resolves to about
% 590 dpi, so it clears the print threshold anyway.
\usepackage{titling}
\pretitle{%
  \begin{center}
  \includegraphics[width=0.55\linewidth]{content/branding/bme-logo.png}\\[2.2em]
  \LARGE}
\posttitle{\end{center}\vskip 0.5em}

% The robot bug, which the brand guidelines describe as an extractable element
% usable as a design device. It sits in the running footer, set at cap height so
% it reads as part of the line rather than as a picture dropped into it. Defined
% here because render/_version.tex, which builds the footer, is included after
% this file.
\newcommand{\bmebug}{%
  \raisebox{-0.35ex}{\includegraphics[height=1.9ex]{content/branding/bme-bug.png}}}

% --- figures ----------------------------------------------------------------
\usepackage{graphicx}
\setkeys{Gin}{width=0.85\linewidth,keepaspectratio}

% Keep every figure exactly where it was written. LaTeX would otherwise float it
% to wherever it fits best, which is right for a book of prose and wrong here:
% these are step-by-step instructions, and a screenshot that says "click this"
% has to sit beside the step that says it. The text says "the image below" in a
% good many places, and with floats enabled that was often no longer true.
\usepackage{float}
\floatplacement{figure}{H}

% --- callouts ---------------------------------------------------------------
% Notion callouts arrive as blockquotes; give them a light shaded box so they
% read as asides rather than as quotations.
\usepackage{framed}
\usepackage{xcolor}
\definecolor{shadecolor}{RGB}{242,244,247}
\let\oldquote\quote
\let\endoldquote\endquote
\renewenvironment{quote}{\begin{snugshade}\begin{oldquote}}%
                        {\end{oldquote}\end{snugshade}}

% --- block reference entries --------------------------------------------------
% The block reference is a lookup chapter, written as a definition list: each
% block signature is the term, what it does is the definition. LaTeX's default
% puts the definition on the same line as the term, which for signatures as long
% as "count with [i] from (n) to (n) by step (n) repeat" leaves the description
% starting somewhere in the middle of a line and gives the eye nothing to scan
% down. nextline puts every signature on a line of its own.
%
% This restyles description lists throughout the document; at present the block
% reference is the only chapter that uses one.
\usepackage{enumitem}
\setlist[description]{style=nextline, leftmargin=1.2em, itemsep=0.35em}

% --- cross-references --------------------------------------------------------
% These materials are meant to be printable, so an internal cross-reference has
% to work on paper as well as under a mouse. tools/xref.lua appends \pageofref to
% every internal link; varioref then phrases it the way a book does, saying "on
% this page" or "on the next page" where that is true and giving a number only
% when the target is further off.
%
% \unskip because the filter emits this immediately after the link text, and the
% starred \vpageref* because the unstarred form supplies its own leading space --
% which would land inside the parenthesis and give "( on page 21)".
\usepackage{varioref}
\newcommand{\pageofref}[1]{\unskip\ (\vpageref*{#1})}
% A reference to a figure carries its number too, so that prose which says "the
% graph below" can name the figure instead. LaTeX supplies the number, so it
% stays right when figures are added or moved.
% No space before \vpageref here: the unstarred form supplies its own, and adding
% one as well loses it -- the comma ends up jammed against "on page".
\newcommand{\figref}[1]{\unskip\ (Figure~\ref{#1},\vpageref{#1})}

% --- long URLs --------------------------------------------------------------
% Allow breaks anywhere in a URL rather than letting it run into the margin.
\usepackage{xurl}

% --- lines that will not fit --------------------------------------------------
% This document is full of things TeX cannot break: file names, block names such
% as when_mBot_starts_up, standards codes, words like function/dysfunction. When
% one lands near the end of a line, TeX would rather run into the margin than
% loosen the spacing. \emergencystretch lets it make one further pass with more
% give, which absorbs the small cases; the large ones have to be fixed in the
% text. tools/check-overfull.sh reports whatever is left.
\emergencystretch=4em

% --- table of contents ------------------------------------------------------
% report.cls sizes the number and page-number boxes for single digits. With ten
% or more chapters this collides: "10Sonar", "8.10Adding color sensor
% extension", and a bold three-digit page number such as 130 overflows its box
% and breaks the right-hand alignment. Widen all three.
\makeatletter
\renewcommand\@pnumwidth{2.6em}
\renewcommand*\l@section{\@dottedtocline{1}{1.5em}{3.2em}}
\renewcommand*\l@subsection{\@dottedtocline{2}{4.7em}{4.1em}}
\renewcommand*\l@chapter[2]{%
  \ifnum \c@tocdepth >\m@ne
    \addpenalty{-\@highpenalty}%
    \vskip 1.0em \@plus\p@
    \setlength\@tempdima{2.8em}%
    \begingroup
      \parindent \z@ \rightskip \@pnumwidth
      \parfillskip -\@pnumwidth
      \leavevmode \bfseries
      \advance\leftskip\@tempdima
      \hskip -\leftskip
      #1\nobreak\hfil \nobreak\hb@xt@\@pnumwidth{\hss #2}\par
      \penalty\@highpenalty
    \endgroup
  \fi}
\makeatother

% keep it out of the PDF's Author field, which should name the institution.

% include-in-header content, so \hypersetup is not yet defined at this point.
\AtBeginDocument{\hypersetup{pdfauthor={University of Cincinnati}}}
